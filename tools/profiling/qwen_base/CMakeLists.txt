cmake_minimum_required(VERSION 3.24)
project(qwen_base_profile LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The native observer requires macOS Metal/Objective-C")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/observer.mm" PROFILE_SOURCE_SHA)
add_library(qwen_base_profile SHARED observer.mm)
target_compile_options(qwen_base_profile PRIVATE -O2 -Wall -Wextra -Werror -fobjc-arc)
target_compile_definitions(qwen_base_profile PRIVATE TC_PROFILE_SOURCE_SHA="${PROFILE_SOURCE_SHA}")
target_link_libraries(qwen_base_profile PRIVATE "-framework Foundation" "-framework Metal" "-framework QuartzCore")
